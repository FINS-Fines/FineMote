#include <type_traits>
#include <iostream>

// 1. 主模板（只做声明，不定义；或者定义一个“兜底”版本）
template <typename T, typename Enable = void>
class Processor; // 未定义，用于阻止不满足条件的类型

// 2. 偏特化 A：仅当 T 是整数时启用
template <typename T>
class Processor<T, std::enable_if_t<std::is_integral_v<T>>> {
public:
    static void process() { std::cout << "处理整数" << std::endl; }
};

// 3. 偏特化 B：仅当 T 是浮点数时启用
template <typename T>
class Processor<T, std::enable_if_t<std::is_floating_point_v<T>>> {
public:
    static void process() { std::cout << "处理浮点数" << std::endl; }
};

int main() {
    Processor<int>::process();    // 匹配偏特化 A
    Processor<double>::process(); // 匹配偏特化 B
    // Processor<std::string>::process(); // 编译报错：未定义的主模板（无法实例化）
}
